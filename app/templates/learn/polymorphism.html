Understanding Polymorphism

The magical ability of objects to take many forms while maintaining the same interface

What is Polymorphism?

Polymorphism means "many forms" in Greek. It's the ability of different objects to respond to the same method call in their own unique way, while sharing a common interface.

Dynamic Behavior

Same method, different implementations

Flexible Code

Easy to extend and maintain systems

Loose Coupling

Reduces dependencies between classes

Interactive Code Examples

Python Polymorphism 
# Base class
class Shape:
    def draw(self):
        pass
    
    def area(self):
        pass

# Polymorphic implementations
class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def draw(self):
        return f"Drawing a circle with radius {self.radius}"
    
    def area(self):
        return 3.14159 * self.radius ** 2

class Square(Shape):
    def __init__(self, side):
        self.side = side
    
    def draw(self):
        return f"Drawing a square with side {self.side}"
    
    def area(self):
        return self.side ** 2

class Triangle(Shape):
    def __init__(self, base, height):
        self.base = base
        self.height = height
    
    def draw(self):
        return f"Drawing a triangle with base {self.base}"
    
    def area(self):
        return 0.5 * self.base * self.height

# Polymorphism in action
shapes = [Circle(5), Square(4), Triangle(3, 6)]

for shape in shapes:
    print(shape.draw())  # Different behavior for each!
    print(f"Area: {shape.area()}")
    print("---")
Java Polymorphism 
// Base class
abstract class Shape {
    public abstract String draw();
    public abstract double area();
}

// Polymorphic implementations
class Circle extends Shape {
    private double radius;
    
    public Circle(double radius) {
        this.radius = radius;
    }
    
    @Override
    public String draw() {
        return "Drawing a circle with radius " + radius;
    }
    
    @Override
    public double area() {
        return Math.PI * radius * radius;
    }
}

class Square extends Shape {
    private double side;
    
    public Square(double side) {
        this.side = side;
    }
    
    @Override
    public String draw() {
        return "Drawing a square with side " + side;
    }
    
    @Override
    public double area() {
        return side * side;
    }
}

// Polymorphism in action
public class PolymorphismDemo {
    public static void main(String[] args) {
        Shape[] shapes = {
            new Circle(5),
            new Square(4)
        };
        
        for (Shape shape : shapes) {
            System.out.println(shape.draw()); // Polymorphic behavior
            System.out.println("Area: " + shape.area());
            System.out.println("---");
        }
    }
}
C++ Polymorphism 
#include <iostream>
#include <vector>
#include <memory>
#include <string>

// Base class with virtual functions
class Shape {
public:
    virtual ~Shape() = default;
    virtual std::string draw() = 0;
    virtual double area() = 0;
};

// Polymorphic implementations
class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(double r) : radius(r) {}
    
    std::string draw() override {
        return "Drawing a circle with radius " + std::to_string(radius);
    }
    
    double area() override {
        return 3.14159 * radius * radius;
    }
};

class Square : public Shape {
private:
    double side;
    
public:
    Square(double s) : side(s) {}
    
    std::string draw() override {
        return "Drawing a square with side " + std::to_string(side);
    }
    
    double area() override {
        return side * side;
    }
};

// Polymorphism in action
int main() {
    std::vector<std::unique_ptr<Shape>> shapes;
    shapes.push_back(std::make_unique<Circle>(5));
    shapes.push_back(std::make_unique<Square>(4));
    
    for (auto& shape : shapes) {
        std::cout << shape->draw() << std::endl;
        std::cout << "Area: " << shape->area() << std::endl;
        std::cout << "---" << std::endl;
    }
    
    return 0;
}

using System;
using System.Collections.Generic;

// Base class
public abstract class Shape 
{
    public abstract string Draw();
    public abstract double Area();
}

// Polymorphic implementations
public class Circle : Shape 
{
    private double radius;
    
    public Circle(double radius) 
    {
        this.radius = radius;
    }
    
    public override string Draw() 
    {
        return $"Drawing a circle with radius {radius}";
    }
    
    public override double Area() 
    {
        return Math.PI * radius * radius;
    }
}

public class Square : Shape 
{
    private double side;
    
    public Square(double side) 
    {
        this.side = side;
    }
    
    public override string Draw() 
    {
        return $"Drawing a square with side {side}";
    }
    
    public override double Area() 
    {
        return side * side;
    }
}

// Polymorphism in action
class Program 
{
    static void Main() 
    {
        List<Shape> shapes = new List<Shape> 
        {
            new Circle(5),
            new Square(4)
        };
        
        foreach (Shape shape in shapes) 
        {
            Console.WriteLine(shape.Draw()); // Polymorphic behavior
            Console.WriteLine($"Area: {shape.Area()}");
            Console.WriteLine("---");
        }
    }
}
