🎨 Understanding Abstraction

🚗 The Car Dashboard Analogy

When you drive a car, you interact with simple controls: steering wheel, pedals, gear shift. You don't need to know about the engine internals, transmission mechanics, or fuel injection systems. The complex implementation is hidden behind a simple interface!

🎛️ Simple Interface:
🎛️ Steering Wheel: Turn left/right
🦶 Gas Pedal: Speed up
🦶 Brake Pedal: Slow down
⚙️ Gear Shift: Change gears
🔑 Ignition: Start/stop engine

Hidden Complexity: Engine management, transmission control, fuel injection, braking systems, electrical systems... 

// Abstraction in Java
public abstract class Vehicle {
    // Abstract methods - must be implemented by subclasses
    public abstract void start();
    public abstract void stop();
    public abstract void accelerate();
    public abstract void brake();
    
    // Concrete method - common to all vehicles
    public void displayInfo() {
        System.out.println("This is a vehicle");
    }
}

// User doesn't need to know HOW each vehicle starts
// They just call start() and it works!

public class Car extends Vehicle {
    @Override
    public void start() {
        // Complex engine starting process hidden here
        igniteEngine();
        initializeSystems();
        warmupEngine();
        System.out.println("Car started!");
    }
    
    private void igniteEngine() { /* complex logic */ }
    private void initializeSystems() { /* complex logic */ }
    private void warmupEngine() { /* complex logic */ }
    
    // ... other implementations
}

🏗️ Abstract Classes

  • • Can have abstract and concrete methods
  • • Can have instance variables
  • • Can have constructors
  • • Cannot be instantiated
  • • Use 'abstract' keyword
abstract class Shape {
  abstract void draw();
  void display() { /* concrete */ }
}

🔌 Interfaces

  • • All methods are abstract (before Java 8)
  • • All variables are public, static, final
  • • No constructors
  • • Pure abstraction
  • • Use 'interface' keyword
interface Drawable {
  void draw();
  void resize();
}

🎮 Complete Game Character Example

// Abstract base class
public abstract class GameCharacter {
    // Instance variables (common to all characters)
    protected String name;
    protected int health;
    protected int level;
    protected int experience;
    
    // Constructor
    public GameCharacter(String name, int health, int level) {
        this.name = name;
        this.health = health;
        this.level = level;
        this.experience = 0;
    }
    
    // Abstract methods - must be implemented by subclasses
    public abstract void attack();
    public abstract void defend();
    public abstract void useSpecialAbility();
    
    // Concrete methods - shared implementation
    public void takeDamage(int damage) {
        health -= damage;
        System.out.println(name + " takes " + damage + " damage. Health: " + health);
        if (health <= 0) {
            die();
        }
    }
    
    public void gainExperience(int exp) {
        experience += exp;
        System.out.println(name + " gains " + exp + " experience!");
        if (experience >= level * 100) {
            levelUp();
        }
    }
    
    public void levelUp() {
        level++;
        health += 20;
        experience = 0;
        System.out.println(name + " leveled up to level " + level + "!");
    }
    
    private void die() {
        System.out.println(name + " has died!");
    }
    
    // Getters
    public String getName() { return name; }
    public int getHealth() { return health; }
    public int getLevel() { return level; }
}

// Concrete implementation 1
public class Warrior extends GameCharacter {
    private int strength;
    private String weapon;
    
    public Warrior(String name, int health, int level, String weapon) {
        super(name, health, level);
        this.strength = level * 10;
        this.weapon = weapon;
    }
    
    @Override
    public void attack() {
        int damage = strength + (int)(Math.random() * 20);
        System.out.println(name + " swings " + weapon + " for " + damage + " damage!");
    }
    
    @Override
    public void defend() {
        System.out.println(name + " raises shield, reducing incoming damage by 50%!");
    }
    
    @Override
    public void useSpecialAbility() {
        System.out.println(name + " uses BERSERKER RAGE! Attack power doubled!");
        strength *= 2;
    }
}

// Concrete implementation 2
public class Mage extends GameCharacter {
    private int mana;
    private String spellBook;
    
    public Mage(String name, int health, int level, String spellBook) {

        this.mana = level * 15;
        this.spellBook = spellBook;

    public void attack() {
        if (mana >= 10) {
            mana -= 10;

            System.out.println(name + " casts fireball for " + damage + " damage! Mana: " + mana);
        } else {
            System.out.println(name + " is out of mana! Staff attack for 5 damage.");
        }
    }
    
    @Override
    public void defend() {
        System.out.println(name + " casts magic shield, absorbing next attack!");
    }
    
    @Override
    public void useSpecialAbility() {
        if (mana >= 30) {
            mana -= 30;

            System.out.println(name + " doesn't have enough mana for special ability!");
        }

    public void restoreMana() {

    }
}

}

🔌 Complete Interface Example

// Interface definitions
public interface Drawable {
    void draw();
    void resize(int width, int height);
    void setColor(String color);
}

public interface Movable {
    void move(int x, int y);
    void rotate(int degrees);
    int getX();
    int getY();
}

public interface Clickable {
    void onClick();
    void onDoubleClick();
    boolean isClickable();
}

// Class implementing multiple interfaces
public class Button implements Drawable, Clickable {
    private String text;
    private String color;
    private int x, y, width, height;
    private boolean enabled;
    
    public Button(String text, int x, int y) {
        this.text = text;
        this.x = x;
        this.y = y;
        this.width = 100;
        this.height = 30;
        this.color = "blue";
        this.enabled = true;
    }
    
    // Implementing Drawable interface
    @Override
    public void draw() {
        System.out.println("Drawing " + color + " button '" + text + 
                          "' at (" + x + "," + y + ") size " + width + "x" + height);
    }
    
    @Override
    public void resize(int width, int height) {
        this.width = width;
        this.height = height;
        System.out.println("Button resized to " + width + "x" + height);
    }
    
    @Override
    public void setColor(String color) {
        this.color = color;
        System.out.println("Button color changed to " + color);
    }
    
    // Implementing Clickable interface
    @Override
    public void onClick() {
        if (enabled) {
            System.out.println("Button '" + text + "' clicked!");
            performAction();
        } else {
            System.out.println("Button is disabled");
        }
    }
    
    @Override
    public void onDoubleClick() {
        if (enabled) {
            System.out.println("Button '" + text + "' double-clicked!");
            performSpecialAction();
        }
    }
    
    @Override
    public boolean isClickable() {
        return enabled;
    }
    
    // Button-specific methods
    private void performAction() {
        System.out.println("Performing button action...");
    }
    
    private void performSpecialAction() {
        System.out.println("Performing special button action...");
    }
    
    public void setEnabled(boolean enabled) {
        this.enabled = enabled;
        System.out.println("Button " + (enabled ? "enabled" : "disabled"));
    }
}

// Another class implementing different interfaces
public class GameSprite implements Drawable, Movable {
    private String name;
    private String color;
    private int x, y, width, height;
    
    public GameSprite(String name, int x, int y) {
        this.name = name;
        this.x = x;
        this.y = y;
        this.width = 50;
        this.height = 50;
        this.color = "red";
    }
    
    // Implementing Drawable interface
    @Override
    public void draw() {
        System.out.println("Drawing " + color + " sprite '" + name + 
                          "' at (" + x + "," + y + ")");
    }
    
    @Override
    public void resize(int width, int height) {
        this.width = width;
        this.height = height;
        System.out.println("Sprite resized to " + width + "x" + height);
    }
    
    @Override
    public void setColor(String color) {
        this.color = color;
        System.out.println("Sprite color changed to " + color);
    }
    
    // Implementing Movable interface
    @Override
    public void move(int newX, int newY) {
        System.out.println("Moving " + name + " from (" + x + "," + y + 
                          ") to (" + newX + "," + newY + ")");
        this.x = newX;
        this.y = newY;
    }
    
    @Override
    public void rotate(int degrees) {
        System.out.println("Rotating " + name + " by " + degrees + " degrees");
    }
    
    @Override
    public int getX() { return x; }
    
    @Override
    public int getY() { return y; }
}

// Usage example
public class InterfaceDemo {
    public static void main(String[] args) {
        // Polymorphism with interfaces
        Drawable[] drawables = {
            new Button("OK", 10, 10),
            new GameSprite("Player", 100, 100),
            new Button("Cancel", 10, 50)
        };
        
        // Draw all drawable objects
        for (Drawable drawable : drawables) {
            drawable.draw();  // Each implements draw() differently
            drawable.setColor("green");
            drawable.resize(120, 40);
        }
        
        // Test clickable objects
        Button okButton = new Button("OK", 10, 10);
        okButton.onClick();
        okButton.onDoubleClick();
        okButton.setEnabled(false);
        okButton.onClick();  // Won't work - disabled
    }
}
✅ Interface Benefits:
  • • Multiple inheritance support
  • • Contract enforcement
  • • Loose coupling
  • • Polymorphism enabler
🔄 When to Use:
  • • Need multiple inheritance
  • • Define contracts
  • • Unrelated classes sharing behavior
  • • Pure abstraction needed
