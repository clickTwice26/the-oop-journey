Understanding Encapsulation

The cornerstone of object-oriented programming that keeps your data safe and your code organized

What is Encapsulation?

Encapsulation is like a protective capsule around your data. It's the practice of bundling data (attributes) and methods (functions) together in a single unit called a class, while restricting direct access to some of the object's components.

Data Protection

Prevents unauthorized access and modification

Controlled Interface

Provides specific methods to interact with data

Modularity

Creates self-contained, reusable components

Real-World Analogy

Python Access Modifiers
class BankAccount:
    def __init__(self, account_number, initial_balance):
        self.account_number = account_number    # Public attribute
        self._balance = initial_balance         # Protected attribute (convention)
        self.__pin = "1234"                    # Private attribute (name mangling)
    
    # Public method - accessible from anywhere
    def get_account_info(self):
        return f"Account: {self.account_number}"
    
    # Protected method - intended for subclasses
    def _validate_transaction(self, amount):
        return amount > 0 and amount <= self._balance
    
    # Private method - only accessible within this class
    def __encrypt_pin(self, pin):
        return f"encrypted_{pin}"
    
    # Public interface for balance access
    def get_balance(self):
        return self._balance
    
    def deposit(self, amount):
        if self._validate_transaction(amount):
            self._balance += amount
            return True
        return False

# Usage example
account = BankAccount("123456", 1000)

# Public access - works fine
print(account.get_account_info())  # "Account: 123456"
print(account.account_number)      # "123456"

# Protected access - works but not recommended
print(account._balance)            # 1000

# Private access - this will cause an AttributeError
# print(account.__pin)             # AttributeError!
Java Access Modifiers
public class BankAccount {
    public String accountNumber;        // Public - accessible everywhere
    protected double balance;           // Protected - accessible in package and subclasses
    private String pin;                 // Private - only within this class
    
    public BankAccount(String accountNumber, double initialBalance) {
        this.accountNumber = accountNumber;
        this.balance = initialBalance;
        this.pin = "1234";
    }
    
    // Public method - accessible from anywhere
    public String getAccountInfo() {
        return "Account: " + accountNumber;
    }
    
    // Protected method - accessible in subclasses
    protected boolean validateTransaction(double amount) {
        return amount > 0 && amount <= balance;
    }
    
    // Private method - only within this class
    private String encryptPin(String pin) {
        return "encrypted_" + pin;
    }
    
    // Public getter for private data
    public double getBalance() {
        return balance;
    }
    
    public boolean deposit(double amount) {
        if (validateTransaction(amount)) {
            balance += amount;
            return true;
        }
        return false;
    }
}
